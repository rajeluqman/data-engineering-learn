      take(old.guided, 'guided', sets.guided, v => v.answered === true, v => v.done === true);
      take(old.questions, 'question', sets.questions, v => v.answered === true, v => v.done === true);
      take(old.labs, 'lab', sets.labs, v => v.touched === true, v => v.touched === true);
      const items = isObject(old.senior?.items) ? old.senior.items : {};
      for (const [id, value] of Object.entries(items)) {
        if (sets.senior.has(id) && isObject(value) &&
            ['review', 'complete'].includes(value.status) &&
            (!own(state.senior, id) ||
             (value.status === 'complete' && state.senior[id].status !== 'complete'))) {
          state.senior[id] = {status: value.status, updatedAt: stamp};
          changed = true;
        }
      }
      // Legacy Senior "last" stores a tab, not a visited card.
      if (!state.legacyMigration) state.legacyMigration = {version: 1, completedAt: stamp};
      if (changed) persist();
      return state;
    }
    function snapshot() { return clone(state); }
    function summary() {
      const count = (array, f) => array.filter(f).length;
      const seniorIds = [...sets.senior];
      return {
        modules: {read: count(catalog.modules, id => own(state.read, 'sql-' + id)), total: catalog.modules.length},
        guided: {completed: count(catalog.guided, id => state.attempts['guided:' + id]?.completed === true),
          total: catalog.guided.length},
        questions: {completed: count(catalog.questions, id => state.attempts['question:' + id]?.completed === true),
          attempted: count(catalog.questions, id => (state.attempts['question:' + id]?.attemptCount || 0) > 0),
          total: catalog.questions.length},
        labs: {touched: count(catalog.labs, id => state.attempts['lab:' + id]?.completed === true),
          total: catalog.labs.length},
        senior: {completed: count(seniorIds, id => state.senior[id]?.status === 'complete'),
          review: count(seniorIds, id => state.senior[id]?.status === 'review'),
          total: seniorIds.length}
      };
    }
    function location(id, stamp) {
      if (!validContent(id, sets)) throw new Error('Unknown content ID');
      const visitedAt = stamp || now();
      if (!validTime(visitedAt)) throw new Error('Invalid visit time');
      state.lastLocation = {contentId: id, section: 'module', visitedAt};
      persist();
      return snapshot();
    }
    function setModuleRead(id, checked) {
      if (!sets.modules.has(id)) throw new Error('Unknown module ID');
      const key = 'sql-' + id;
      if (checked) state.read[key] = {markedAt: now(), selfReported: true};
      else delete state.read[key];
      // Keep the older page writer in step so a later migration cannot undo an unmark.
      if (storage) {
        const legacy = readJson(storage, LEGACY.modules, warnings) || {};
        legacy[id] = !!checked;
        try { storage.setItem(LEGACY.modules, JSON.stringify(legacy)); }
        catch (_) { warnings.push('Legacy module state could not be saved'); }
      }
      persist();
      return snapshot();
    }
    function resume() {
      const last = state.lastLocation;
      if (last && validContent(last.contentId, sets)) {
        const id = last.contentId.slice(4);
        return {kind: 'resume', contentId: last.contentId,
          href: BASE + '/sql-learning-path#' + id.toLowerCase(), label: id};
      }
      const next = catalog.modules.find(id => !own(state.read, 'sql-' + id));
      return next ? {kind: 'start', contentId: 'sql-' + next,
        href: BASE + '/sql-learning-path#' + next.toLowerCase(), label: next} :
        {kind: 'practice', contentId: null, href: BASE + '/sql-field-lab-full',
          label: 'SQL Assessment Bank'};
    }
    function exportState() { return JSON.stringify(state, null, 2); }
    function importState(json) {
      const parsed = JSON.parse(json);
      const replacement = safeState(parsed, sets, true);
      state = replacement;
      persist();
      return snapshot();
    }
    function reset() {
      state = empty();
      if (persistent) {
        try {
          storage.removeItem(KEY);
          for (const key of Object.values(LEGACY)) storage.removeItem(key);
        }
        catch (_) { persistent = false; warnings.push('Progress reset in memory only'); }
      }
      return snapshot();
    }
    function subscribeStorage(onChange) {
      if (typeof root.addEventListener !== 'function') return () => {};
      const listener = event => {
        if (Object.values(LEGACY).includes(event.key)) {
          migrate();
          onChange(snapshot());
          return;
        }
        if (event.key !== KEY) return;
        try {
          state = event.newValue ? safeState(JSON.parse(event.newValue), sets, false) : empty();
          onChange(snapshot());
        } catch (_) { warnings.push('Other tab sent invalid progress'); }
      };
      root.addEventListener('storage', listener);
      return () => root.removeEventListener('storage', listener);
    }
    return {migrate, snapshot, summary, location, setModuleRead, resume, exportState,
      importState, reset, subscribeStorage, warnings, get persistent() { return persistent; }};
  }

  return {KEY, LEGACY, create, safeState};
});
